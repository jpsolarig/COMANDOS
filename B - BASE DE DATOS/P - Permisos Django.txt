
### VER PERMISOS ###

### CORE ###

SELECT id, name, codename
FROM auth_permission
WHERE content_type_id IN (
    SELECT id FROM django_content_type WHERE app_label = 'core'
)
ORDER BY codename;

### PERSONA ###

SELECT id, name, codename
FROM auth_permission
WHERE content_type_id IN (
    SELECT id FROM django_content_type WHERE app_label = 'persona'
)
ORDER BY codename;
